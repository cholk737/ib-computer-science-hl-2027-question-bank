def house_sort(entries):
    result = [item for item in entries if item is not None]
    for end in range(len(result)-1, 0, -1):

def follow_all(links, limit):
    success = [False] * len(links)
    for target in range(len(links)):
        index = target
        for attempt in range(limit):

            if value == target:
                success[target] = True

            index = value
    return success

def top_five(products):
    ordered = products[:]
    for i in range(len(ordered)-1):
        best = i
        for j in range(i+1, len(ordered)):
