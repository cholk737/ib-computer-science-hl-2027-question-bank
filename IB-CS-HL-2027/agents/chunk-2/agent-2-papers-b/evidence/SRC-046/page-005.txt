Example 1 
SELECT STAFF.Firstname, STAFF.Surname, Role.PayGrade 
FROM STAFF INNER JOIN Roles ON STAFF.RoleID = ROLE. RoleID 
WHERE STAFF.Surname = ‘Waters’ AND ROLE.PayGrade = 17;

FROM STAFF, ROLE
